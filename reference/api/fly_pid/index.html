    fly_pid¶

    fly_pid ¶

    Classical baseline: flies the same approach, flare and landing as fly.py with cascaded PID loops (FlightGear's algorithm, see pid.py) instead of the networks. SIMULATION ONLY.

    Same UDP link as fly.py (fg_interface.py, 5501 in / 5502 out), same loop with FlightGear (fly.run), same phases (final_approach -> flare -> landing), same ground rules (brakes, controls) and the same 10 Hz command rate. What it does NOT use is the networks' data pipeline (Butterworth filter, report features, z-score): the PID reads FlightGear's raw packets, taking the latest one every 0.1 s.

    CONTROLLER=pid bash scripts/run_flight.sh        # fly_pid.py + FlightGear together
python src/fly_pid.py      # or start this first, then bash scripts/launch_flightgear.sh

    Loops (outer -> inner), all at 10 Hz: lateral, approach cross-track [m] -> heading offset [deg] -> bank [deg] -> aileron sideslip [rad] -> rudder (coordination) lateral, flare sideways speed [fps] -> bank [deg] -> aileron (no drift) heading error [deg] -> rudder (de-crab) longitudinal height above the glide path [ft] -> vertical speed [fps] -> pitch [deg] -> elevator flare: vertical speed reference = -agl / FLARE_TAU_S (exponential flare) speed airspeed [kt] -> throttle (approach); idle in the flare flaps full (1.0) from takeover, as the pilots flew

    The log and the gains used are saved to artifacts/flights/pid_.csv / .json.

    class PIDAutopilot:
    """Drop-in for fly.Autopilot: on_packet(o) -> [(features, command)] at 10 Hz."""

    def __init__(self, gains=GAINS):
        self.next_tick = None
        self.pid = {name: PIDController(g, GRID_S) for name, g in gains.items()}
        self.phase = "final_approach"
        self.wow_since = None
        self.started = False
        self.elevator = 0.0         # held where the flare left it after touchdown

    def on_packet(self, o):
        """Feed one 60 Hz packet; every 0.1 s of sim time, the latest one is used to command."""
        if self.next_tick is not None and o["t"] < self.next_tick - 1e-6:
            return []
        self.next_tick = (self.next_tick or o["t"]) + GRID_S
        if o["t"] >= self.next_tick:            # a gap in the data: restart the grid here
            self.next_tick = o["t"] + GRID_S
        f = features(o)
        return [(f, self.step(f))]

    def take_control(self, f):
        """First tick: seed every loop with where the aircraft is, so nothing jumps."""
        self.pid["vs_to_pitch"].reset(f["pitch_deg"])
        self.pid["speed_to_throttle"].reset(THROTTLE_START)
        self.started = True

    def update_phase(self, f):
        """final approach -> flare -> landing, never back (as fly.Autopilot)."""
        if f["wow"] == 1:
            self.wow_since = self.wow_since if self.wow_since is not None else f["t"]
        else:
            self.wow_since = None
        if self.phase != "landing" and self.wow_since is not None \
                and f["t"] - self.wow_since >= TOUCHDOWN_HOLD_S - 1e-6:
            self.phase = "landing"
        elif self.phase == "final_approach" and f["flare_flag"] == 1:
            self.phase = "flare"
            # the flare loops take over from the approach loops where those left off
            self.pid["drift_to_bank"].reset(self.pid["heading_to_bank"].u_n_1)
            self.pid["heading_to_rudder"].reset(self.pid["slip_to_rudder"].u_n_1)
        f["phase"] = self.phase

    def step(self, f):
        if not self.started:
            self.take_control(f)
        self.update_phase(f)
        p = self.pid

        if self.phase == "landing":
            # on the ground: the same rules as fly.py, so the roll-out does not differ
            return self.command(0.0, self.elevator, 0.0, 0.0, BRAKE_LEVEL if f["gs_kt"] < BRAKE_BELOW_KT else 0.0)

        # ---- lateral
        if self.phase == "final_approach":
            heading_ref = p["xtk_to_heading"](f["cross_track_error_m"], 0.0)
            bank_ref = p["heading_to_bank"](f["heading_error_deg"], heading_ref)
            rudder = p["slip_to_rudder"](f["sideslip_rad"], 0.0)
        else:
            bank_ref = p["drift_to_bank"](f["lateral_body_speed_fps"], 0.0)
            rudder = p["heading_to_rudder"](f["heading_error_deg"], 0.0)
        aileron = p["bank_to_aileron"](f["roll_deg"], bank_ref)

        # ---- longitudinal
        if self.phase == "final_approach":
            distance_m = AIM_POINT_M - f["along_track_m"]
            height_ref_ft = max(distance_m, 0.0) * math.tan(math.radians(GLIDESLOPE_REF_DEG)) / 0.3048
            vs_nominal = -f["gs_kt"] * 1.68781 * math.tan(math.radians(GLIDESLOPE_REF_DEG))
            vs_ref = vs_nominal + p["height_to_vs"](f["height_above_threshold_ft"], height_ref_ft)
            throttle = p["speed_to_throttle"](f["airspeed_kt"], V_REF_KT)
        else:
            vs_ref = -max(f["agl_ft"] / FLARE_TAU_S, FLARE_MIN_SINK_FPS)
            throttle = 0.0
        pitch_ref = p["vs_to_pitch"](f["vertical_speed_fps"], vs_ref)
        self.elevator = p["pitch_to_elevator"](f["pitch_deg"], pitch_ref)

        f.update(bank_ref_deg=bank_ref, pitch_ref_deg=pitch_ref, vs_ref_fps=vs_ref)     # for the log
        return self.command(aileron, self.elevator, rudder, throttle, 0.0)

    @staticmethod
    def command(aileron, elevator, rudder, throttle, brake):
        return dict(aileron=clip(aileron), elevator=clip(elevator), rudder=clip(rudder),
                    throttle=clip(throttle, 0.0, 1.0), flaps=FLAPS, brake=clip(brake, 0.0, 1.0))

    def features(o):
    """What the loops need, straight from one raw FlightGear packet (no filter).
    Named like fly.py's features, so the log and fly.run's summary read the same."""
    along, cross = to_runway_frame(o["lat"], o["lon"])
    return dict(
        t=o["t"],
        along_track_m=along,
        cross_track_error_m=cross,
        heading_error_deg=wrap180(o["heading"] - RUNWAY["heading_deg"]),
        height_above_threshold_ft=o["alt_ft"] - RUNWAY["elevation_ft"],
        agl_ft=o["agl_ft"],
        roll_deg=o["roll"],
        pitch_deg=o["pitch"],
        sideslip_rad=o["side_slip_rad"],
        lateral_body_speed_fps=o["v_lateral_fps"],
        airspeed_kt=o["ias"],
        gs_kt=o["gs_kt"],
        vertical_speed_fps=o["vs_fps"],
        flaps_state_norm=o["flaps"],
        flare_flag=int(o["agl_ft"] <= FLARE_AGL_FT),
        wow=o["wow"],
        elevator_trim=o["elevator_trim"],
        crashed=o["crashed"],
    )
